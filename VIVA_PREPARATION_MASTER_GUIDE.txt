3.1 MATHEMATICAL ENGINE: src/simulators/nfa-dfa/subset-construction.ts

Function 1: computeEpsilonClosure(states: string[], nfa: NFADefinition): string[]
---------------------------------------------------------------------------------
export function computeEpsilonClosure(states: string[], nfa: NFADefinition): string[] {
  // Initialize closure set with the input seed states.
  // Set guarantees O(1) membership check and prevents duplicate state insertions.
  const closure = new Set<string>(states);

  // Initialize FIFO queue for BFS traversal with a shallow copy of input states.
  const queue = [...states];

  // Process states in queue until all reachable epsilon paths are exhausted.
  while (queue.length > 0) {
    // Dequeue front state (FIFO). Non-null assertion '!' is safe due to length check.
    const current = queue.shift()!;

    // Iterate over all transitions defined in NFA.
    for (const t of nfa.transitions) {
      // Check if transition originates from 'current' state AND is an epsilon move.
      // Supports 'epsilon', empty string '', or legacy 'e'.
      if (t.fromState === current && (t.symbol === EPSILON || t.symbol === '' || t.symbol === 'e')) {
        // Cycle Prevention: Only visit target state if not already in closure set.
        // Prevents infinite loops on cyclic transitions: q0 -> epsilon -> q1 -> epsilon -> q0.
        if (!closure.has(t.toState)) {
          closure.add(t.toState);   // Add newly discovered state to closure.
          queue.push(t.toState);    // Enqueue to explore transitive epsilon moves: q0 -> eps -> q1 -> eps -> q2.
        }
      }
    }
  }

  // Convert Set back to an Array and sort alphabetically.
  // Sorting is strictly mandatory so that canonical subset names like '{q0, q1}' are deterministic.
  return Array.from(closure).sort();
}

Time Complexity: O(|Q| * |T|), where each state is enqueued at most once, scanning the transition array T.
Space Complexity: O(|Q|) auxiliary memory for the Set and queue.


Function 2: computeDirectMove(states: string[], symbol: string, nfa: NFADefinition): string[]
---------------------------------------------------------------------------------------------
export function computeDirectMove(states: string[], symbol: string, nfa: NFADefinition): string[] {
  // Set ensures destination states are unique across multiple originating branches.
  const destinations = new Set<string>();

  // Iterate through every state present in current subset S.
  for (const q of states) {
    // Scan NFA transition list.
    for (const t of nfa.transitions) {
      // Match source state and exact symbol read.
      if (t.fromState === q && t.symbol === symbol) {
        destinations.add(t.toState); // Collect target state.
      }
    }
  }

  // Return sorted array of unique destination states.
  return Array.from(destinations).sort();
}


Function 3: convertNfaToDfa(nfa: NFADefinition): SubsetConversionResult
-----------------------------------------------------------------------
export function convertNfaToDfa(nfa: NFADefinition): SubsetConversionResult {
  // Step 1: Start state of DFA is the epsilon-closure of the NFA start state.
  const initialClosure = computeEpsilonClosure([nfa.initialState], nfa);
  const initialSubset = formatSubset(initialClosure); // Formats ['q0', 'q1'] -> "{q0, q1}"

  // Discovery tracker to ensure subsets are processed exactly once.
  const discoveredSubsets: string[] = [initialSubset];
  const subsetMap: Record<string, string[]> = {
    [initialSubset]: initialClosure,
  };

  const queue: string[] = [initialSubset]; // FIFO queue of subsets pending expansion.
  const transitions: DFATransition[] = [];
  const table: Record<string, Record<string, string>> = {};
  const steps: ConversionStep[] = [];
  let stepCounter = 0;

  // Main Loop: Process unexpanded subsets from the queue.
  while (queue.length > 0) {
    const currentSubsetName = queue.shift()!;
    const currentStates = subsetMap[currentSubsetName] || [];

    if (!table[currentSubsetName]) {
      table[currentSubsetName] = {};
    }

    // For each alphabet symbol, compute deterministic transition.
    for (const symbol of nfa.alphabet) {
      stepCounter++;
      let targetSubsetName: string;
      let reachedNFAStates: string[] = [];
      let closureStates: string[] = [];
      let explanation = '';
      let isNewSubset = false;

      // Dead/Trap State Handling
      if (currentSubsetName === TRAP_STATE_NAME) {
        // Dead state (empty set) has fixed self-loops on every symbol to guarantee total DFA completeness.
        targetSubsetName = TRAP_STATE_NAME;
        explanation = `From dead state ${TRAP_STATE_NAME}, reading '${symbol}' remains in ${TRAP_STATE_NAME}.`;
      } else {
        // Standard Step: delta'(S, a) = epsilon-closure(Move(S, a))
        reachedNFAStates = computeDirectMove(currentStates, symbol, nfa);
        closureStates = computeEpsilonClosure(reachedNFAStates, nfa);
        targetSubsetName = formatSubset(closureStates);

        if (closureStates.length === 0) {
          // If no transitions exist in NFA, route deterministically to dead state.
          explanation = `From subset ${currentSubsetName}, on symbol '${symbol}', no NFA transitions are defined. This leads to dead/trap state ${TRAP_STATE_NAME}.`;
        } else {
          explanation = `From subset ${currentSubsetName}, reading '${symbol}' reaches NFA states {${reachedNFAStates.join(', ')}}. Next DFA state is ${targetSubsetName}.`;
        }
      }

      // Discovery of new DFA state.
      if (!discoveredSubsets.includes(targetSubsetName)) {
        isNewSubset = true;
        discoveredSubsets.push(targetSubsetName);
        subsetMap[targetSubsetName] = closureStates;
        queue.push(targetSubsetName); // Enqueue newly discovered subset for future expansion.
        explanation += ` Discovered new DFA state ${targetSubsetName}.`;
      }

      // Record total DFA transition and lookup table entry.
      transitions.push({
        fromSubset: currentSubsetName,
        symbol,
        toSubset: targetSubsetName,
      });
      table[currentSubsetName][symbol] = targetSubsetName;

      // Record discrete step telemetry for interactive UI walkthrough.
      steps.push({
        stepIndex: stepCounter,
        currentSubset: currentSubsetName,
        currentSubsetStates: currentStates,
        symbol,
        reachedNFAStates,
        closureStates,
        targetSubsetName,
        isNewSubset,
        explanation,
        queueSnapshot: [...queue],
        discoveredSubsets: [...discoveredSubsets],
        dfaTransitionsSnapshot: [...transitions],
      });
    }
  }

  // Final States Identification:
  // A DFA subset is accepting if it contains AT LEAST ONE accept state of original NFA.
  const acceptStates = discoveredSubsets.filter(subsetName => {
    if (subsetName === TRAP_STATE_NAME) return false;
    const states = subsetMap[subsetName] || [];
    return states.some(q => nfa.acceptStates.includes(q));
  });

  const dfa: DFADefinition = {
    states: discoveredSubsets,
    subsets: subsetMap,
    alphabet: [...nfa.alphabet],
    initialState: initialSubset,
    acceptStates,
    transitions,
    trapState: discoveredSubsets.includes(TRAP_STATE_NAME) ? TRAP_STATE_NAME : undefined,
  };

  return { dfa, steps, table };
}


3.2 DUAL STRING SIMULATION ENGINE: src/simulators/nfa-dfa/simulator.ts
---------------------------------------------------------------------
Simulates arbitrary input string on both NFA and DFA synchronously.
- NFA advances by tracking a set of active states: activeNFA = epsilonClosure(directMove(activeNFA, char)).
- DFA advances deterministically: activeDFA = table[activeDFA][char].
- Acceptance verification: NFA accepts if activeNFA contains an accept state; DFA accepts if activeDFA is in acceptStates.
- Language Equivalence assertion: nfaAccepted === dfaAccepted.

CHAPTER 4: AUTOMATA THEORY FOUNDATIONS & MATHEMATICAL PROOFS
================================================================================

1. FORMAL DEFINITIONS:
- NFA 5-Tuple: M_NFA = (Q, Sigma, delta, q0, F)
  * Q: Finite set of states
  * Sigma: Alphabet of non-epsilon symbols
  * delta: Q x (Sigma U {epsilon}) -> P(Q) (Nondeterministic transition mapping)
  * q0 in Q: Initial state
  * F subset of Q: Final / accepting states

- DFA 5-Tuple: M_DFA = (Q', Sigma, delta', q0', F')
  * Q' subset of P(Q): States are subsets of NFA states
  * q0' = epsilon-closure({q0})
  * delta': Q' x Sigma -> Q', where delta'(S, a) = epsilon-closure( Union_{q in S} delta(q, a) )
  * F' = { S in Q' | S intersect F_NFA != empty }

2. RABIN-SCOTT THEOREM (1959):
Every language recognized by an NFA can be recognized by an equivalent DFA.
Hence: L(NFA) = L(DFA) = Regular Languages (REG).

3. POWERSET STATE EXPLOSION (2^n BOUND):
If |Q| = n, the theoretical maximum number of DFA states is |P(Q)| = 2^n.
Witness language: L_k = { w in {0,1}* | the k-th symbol from the end is 1 }.
- NFA requires k + 1 states (nondeterministic guess).
- DFA requires at least 2^k states (must memorize last k symbols in state permutations).
This mathematically proves that nondeterminism provides exponential succinctness.

CHAPTER 5: SOFTWARE ENGINEERING & DESIGN DECISIONS
================================================================================
- Set vs Array: Used JavaScript Set for O(1) membership check during BFS epsilon-closure; prevents infinite loops on cycles.
- Canonical String Naming: Sorted strings '{q0, q1}' used because JavaScript objects compare arrays by reference, not value (['q0'] !== ['q0']). String serialization ensures unique hash mapping.
- Pure SVG vs D3.js: Built custom SVG layout with force-relaxation to avoid 500KB bundle overhead, ensuring 60fps interaction and zero third-party framework fragility.
- Total DFA Function: Explicit dead state (empty set) generated with self-loops so that delta' is a mathematically total function for all symbols.

[Category 2: Automata Theory & NFA/DFA Foundations]
11. Q: Formally define an NFA.
    A: An NFA is a 5-tuple M = (Q, Sigma, delta, q0, F) where delta maps Q x (Sigma U {epsilon}) to the power set P(Q).
12. Q: Formally define a DFA.
    A: A DFA is a 5-tuple M = (Q, Sigma, delta, q0, F) where delta is a total function mapping Q x Sigma to exactly one state in Q.
13. Q: What is the principal difference between NFA and DFA?
    A: An NFA permits multiple next states, missing moves, and epsilon transitions; a DFA mandates exactly one deterministic next state for every alphabet symbol.
14. Q: What is an epsilon transition?
    A: A spontaneous transition that changes the automaton's internal state without consuming an input symbol from the tape.
15. Q: What is an epsilon-closure?
    A: The set containing state q along with all states reachable from q by traversing zero or more consecutive epsilon transitions.
16. Q: Does epsilon-closure of q always include q itself?
    A: Yes, because traversing zero epsilon transitions keeps the machine in state q.
17. Q: What is a dead state or trap state?
    A: A non-accepting state from which the automaton cannot escape (all transitions loop back to itself).
18. Q: Why is a dead state necessary in a DFA?
    A: Because by formal definition, a DFA transition function delta: Q x Sigma -> Q must be total (defined for every symbol).
19. Q: What is a regular language?
    A: A formal language recognized by a finite automaton or described by a regular expression.
20. Q: Do NFAs recognize more languages than DFAs?
    A: No. By the Rabin-Scott theorem, both recognize the identical class of Regular Languages.

[Category 3: Subset Construction Algorithm]
21. Q: What is the subset construction algorithm?
    A: A deterministic algorithm that systematically constructs a DFA whose states correspond to subsets of states of an NFA.
22. Q: What is the initial state of the converted DFA?
    A: The epsilon-closure of the NFA initial state: q0' = epsilon-closure({q0}).
23. Q: How is a transition delta'(S, a) computed in the DFA?
    A: By taking the direct move from subset S on symbol a, followed by the epsilon-closure: delta'(S, a) = epsilon-closure( Union_{q in S} delta(q, a) ).
24. Q: How are accept states determined in the converted DFA?
    A: Any subset S containing at least one accept state of the original NFA is marked as accepting in the DFA.
25. Q: What happens when Move(S, a) is empty?
    A: The transition routes to the dead/trap state (empty set), which self-loops on all alphabet symbols.
26. Q: What queue discipline is used to discover subsets?
    A: A FIFO queue implementing Breadth-First Search (BFS) discovery.
27. Q: How does the algorithm terminate?
    A: When the discovery queue becomes empty, meaning all reachable subsets have been fully expanded.
28. Q: What is the maximum number of states in the converted DFA?
    A: If the NFA has n states, the converted DFA has at most 2^n states.
29. Q: Why do practical DFAs rarely reach 2^n states?
    A: Because most theoretical subsets in P(Q) are unreachable from the initial subset q0'.
30. Q: Give an example of a language that triggers the worst-case 2^n state explosion.
    A: The language of binary strings where the k-th symbol from the end is 1 requires 2^k DFA states.

[Category 4: Code Implementation Deep-Dive]
31. Q: How does computeEpsilonClosure prevent infinite loops on cyclic epsilon transitions?
    A: It uses a Set ('closure') and checks 'if (!closure.has(t.toState))' before pushing to the traversal queue.

    A: Objects compare keys by primitive string equality; arrays are compared by object reference, so ['q0'] !== ['q0'].
